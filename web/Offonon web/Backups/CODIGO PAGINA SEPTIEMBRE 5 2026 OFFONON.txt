    <script>
        // Mobile menu toggle
        const mobileMenuButton = document.getElementById('mobile-menu-button');
        const mobileMenu = document.getElementById('mobile-menu');
        
        mobileMenuButton.addEventListener('click', () => {
            mobileMenu.classList.toggle('hidden');
        });

        // Dynamic Form Fields
        function showFields(service) {
            // Hide all fields first
            document.querySelectorAll('.dynamic-field').forEach(field => {
                field.style.display = 'none';
            });

            // Show relevant fields based on selection
            if (service === 'Clases') {
                document.getElementById('clases-fields').style.display = 'block';
            } else if (service === 'Equipment rental') {
                document.getElementById('equipment-fields').style.display = 'block';
            } else {
                document.getElementById('project-fields').style.display = 'block';
            }
        }

        // Custom Cursor
        const cursor = document.querySelector('.cursor');
        const cursorDot = document.querySelector('.cursor-dot');

        document.addEventListener('mousemove', (e) => {
            cursor.style.left = e.clientX + 'px';
            cursor.style.top = e.clientY + 'px';
        });

        document.querySelectorAll('a, button, .btn-primary, .btn-outline, input[type="submit"], .social-icon').forEach(element => {
            element.addEventListener('mouseenter', () => {
                cursor.classList.add('hover');
            });
            element.addEventListener('mouseleave', () => {
                cursor.classList.remove('hover');
            });
        });

        // Scroll Reveal Animations
        function checkScroll() {
            const sections = document.querySelectorAll('section');
            const windowHeight = window.innerHeight;
            const windowTop = window.scrollY;
            const windowBottom = windowTop + windowHeight;

            sections.forEach(section => {
                const sectionTop = section.offsetTop;
                const sectionBottom = sectionTop + section.offsetHeight;

                if (sectionBottom >= windowTop && sectionTop <= windowBottom) {
                    section.classList.add('reveal');

                    // Section-specific animations
                    if (section.classList.contains('hero-section')) {
                        section.classList.add('hero-animated');
                    } else if (section.classList.contains('services-section')) {
                        section.classList.add('services-reveal');
                    } else if (section.classList.contains('equipment-section')) {
                        section.classList.add('equipment-reveal');
                    } else if (section.classList.contains('spotify-section')) {
                        section.classList.add('spotify-reveal');
                    } else if (section.classList.contains('contact-section')) {
                        section.classList.add('contact-reveal');
                    }
                }
            });
        }

        // Form Submission
        document.getElementById('contactForm').addEventListener('submit', function(e) {
            e.preventDefault();
            
            // Simulate form submission
            document.getElementById('form-fields').style.display = 'none';
            document.getElementById('form-success').classList.remove('hidden');
            
            // Reset form after 5 seconds
            setTimeout(() => {
                document.getElementById('form-fields').style.display = 'block';
                document.getElementById('form-success').classList.add('hidden');
                this.reset();
            }, 5000);
        });

        // Initialize
        document.addEventListener('DOMContentLoaded', function() {
            showFields(document.getElementById('service').value);
            
            // Add smooth transition effect
            document.querySelectorAll('.dynamic-field').forEach(field => {
                field.style.transition = 'opacity 0.3s ease, height 0.3s ease';
            });

            // Show social sidebar after delay
            setTimeout(() => {
                document.querySelector('.social-sidebar').classList.add('visible');
            }, 1000);

            // Initial scroll check
            checkScroll();
        });

        // Check scroll on scroll
        window.addEventListener('scroll', checkScroll);
        window.addEventListener('resize', checkScroll);

        // Lazy loading for images
        if ('loading' in HTMLImageElement.prototype) {
            const images = document.querySelectorAll('img[loading="lazy"]');
            images.forEach(img => {
                img.loading = 'lazy';
            });
        } else {
            // Fallback for browsers that don't support loading="lazy"
            const lazyLoad = function() {
                const lazyImages = document.querySelectorAll('img[loading="lazy"]');
                
                lazyImages.forEach(img => {
                    if (img.getBoundingClientRect().top < window.innerHeight + 200) {
                        img.src = img.dataset.src;
                        img.removeAttribute('loading');
                    }
                });
            };

            document.addEventListener('scroll', lazyLoad);
            window.addEventListener('resize', lazyLoad);
            window.addEventListener('orientationchange', lazyLoad);
            lazyLoad();
        }
    </script>
